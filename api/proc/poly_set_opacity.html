    subroutine poly_set_opacity(self, opacity)
        class(Polyhedron), intent(inout) :: self
        real(wp), intent(in) :: opacity
        self%opacity_ = max(0.0_wp, min(1.0_wp, opacity))
        call make_polyhedron(self)
    end subroutine poly_set_opacity
