    pure subroutine append(a, b)
        real(wp), allocatable, intent(inout) :: a(:,:)
        real(wp), intent(in) :: b(:,:)
        real(wp), allocatable :: tmp(:,:)
        if (.not. allocated(a)) allocate(a(3, 0))
        allocate(tmp(3, size(a, 2) + size(b, 2)))
        tmp(:, 1:size(a, 2)) = a
        tmp(:, size(a, 2)+1:) = b
        call move_alloc(tmp, a)
    end subroutine append
