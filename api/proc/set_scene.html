    subroutine set_scene(self, fm)
        class(WindowProxy), intent(inout) :: self
        type(FrameManager), intent(in), target :: fm
        call forget_handles(self)
        self%scene => fm
        self%scene_seen = -1
        self%scene_added = .false.
    end subroutine set_scene
