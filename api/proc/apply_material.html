    subroutine apply_material(self)
        type(Sphere), intent(inout) :: self
        integer :: k
        call apply_material_to(self, self%mesh)
        do k = 1, size(self%lod_mesh)
            call apply_material_to(self, self%lod_mesh(k))
        end do
