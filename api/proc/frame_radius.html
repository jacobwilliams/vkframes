    real(wp) function frame_radius(frame) result(r)
        class(ReferenceFrame), intent(in) :: frame
        real(wp) :: radii(3)
        r = 0.0_wp
        select type (frame)
        class is (Sphere)
            radii = frame%radii()
            r = max(radii(1), radii(2))
        class is (TerrainBody)
            r = frame%body_radius()
        end select
    end function frame_radius
