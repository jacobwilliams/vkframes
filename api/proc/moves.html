    recursive logical function moves(f) result(m)
        class(ReferenceFrame), intent(in) :: f              !! the frame
        class(ReferenceFrame), pointer :: g
        m = f%is_following()
        if (m .or. f%num_parents() == 0) return
        g => f%parent(1)
        m = moves(g)
    end function moves
