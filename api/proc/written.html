written

private subroutine written(self, i1, i2, x)

Points i1 .. i2 were written: measure a strip's distances along it (from the distance at point i1-1), update its level of detail, and upload the chunks holding them. x: the points in double precision, x(:, i) for chain point i (at least i1-1 .. i2), to measure from; without it, from the floats written. (i2 is the last point written in the chain.)

    subroutine written(self, i1, i2, x)
        class(LineChain), intent(inout) :: self
        integer, intent(in) :: i1, i2
        real(wp), intent(in), optional :: x(:,:)        !! (3, at least i1-1 .. i2): the points
        integer :: j, lo, hi
        do j = 1, self%n_chunks
            associate (c => self%chunks(j))
                lo = max(i1, c%first)
                hi = min(i2, c%first + c%capacity - 1)
                if (lo > hi) cycle
                if (self%kind == CHAIN_STRIP) then
                    call measure(self, j, lo, hi, x)
                    call update_lod(self, j, lo, hi, x)
                end if
                call check_status(vkf_line_points_changed(c%line, int(hi - c%first + 1, c_int)), 'trajectory artists')
            end associate
        end do
    end subroutine written
