private function new_radial_plane(name, color) result(p)

A radial plane named name (decorations of color color): radius 1, circles every 0.5, radial lines every 30 degrees, light gray and transparent (OpenFrames' defaults).

    function new_radial_plane(name, color) result(p)
        character(len=*), intent(in) :: name
        real(wp), intent(in), optional :: color(:)
        type(RadialPlane) :: p
        p%ReferenceFrame = ReferenceFrame(name, color)
        call place_axes(p, p%radius_)
    end function new_radial_plane
