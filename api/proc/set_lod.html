Level of detail on (the default: seen from far away, a strip is drawn from fewer points, within half a pixel of all of them) or off (every point drawn).

    subroutine set_lod(self, enabled)
        class(LineChain), intent(inout) :: self
        logical, intent(in) :: enabled
        integer :: j
        self%lod = enabled
        if (self%kind /= CHAIN_STRIP) return
        do j = 1, self%n_chunks
            call send_lod(self, j)
        end do
    end subroutine set_lod
