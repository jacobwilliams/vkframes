    pure logical function data_valid(self)
        class(TrajectoryFollower), intent(in) :: self
        integer :: k
        data_valid = .true.
        do k = 1, self%n
            if (.not. self%trajs(k)%p%verify_data(self%sources_)) data_valid = .false.
        end do
    end function data_valid
