    function current_view(self) result(v)
        class(WindowProxy), intent(in), target :: self
        type(View), pointer :: v
        if (self%current >= 1 .and. self%current <= self%n_views) then
            v => self%views(self%current)%p
        else
            v => self%default_view
        end if
    end function current_view
