    subroutine sort_unique(a, n)
        real(wp), intent(inout) :: a(:)                     !! the values (1:n used)
        integer, intent(inout) :: n                         !! the number of values
        integer :: i, m
        call quicksort(a(1:n))
        m = min(n, 1)
        do i = 2, n
            if (a(i) /= a(m)) then
                m = m + 1
                a(m) = a(i)
            end if
        end do
        n = m
    end subroutine sort_unique
