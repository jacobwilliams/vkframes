    function frame_to_root(self, f) result(m)
        type(View), intent(in) :: self
        class(ReferenceFrame), intent(in) :: f
        real(wp) :: m(4,4)
        m = f%world_matrix()
        if (associated(self%root)) m = matmul(rigid_inverse(self%root%world_matrix()), m)
    end function frame_to_root
