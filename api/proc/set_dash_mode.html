set_dash_mode

private subroutine set_dash_mode(self, mode, length)

For lines, how the dashes are laid out: 0 anchored to the line (the default), 1 screen-locked (a constant length in pixels, following the line on screen), 2 a fixed length of the line (strips): length scene units for the pattern, or if 0 its length on screen when first drawn.

    subroutine set_dash_mode(self, mode, length)
        class(LineChain), intent(inout) :: self
        integer, intent(in) :: mode
        real(wp), intent(in) :: length
        integer :: j
        self%dash_mode = mode
        self%dash_length = length
        do j = 1, self%n_chunks
            call apply_style(self, self%chunks(j)%line)
        end do
    end subroutine set_dash_mode
