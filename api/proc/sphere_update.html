private recursive subroutine sphere_update(self, t)

Update the frame (followed trajectories, children) and, with level of detail, build the mesh the views have wanted if it isn't built yet (the coarsest with as many segments as wanted, or the finest).

    recursive subroutine sphere_update(self, t)
        class(Sphere), intent(inout), target :: self
        real(wp), intent(in) :: t                           !! the scene time
        integer :: wanted, k
        call self%ReferenceFrame%update(t)
        if (.not. c_associated(self%lod_node)) return
        wanted = vkf_sphere_lod_requested(self%lod_node)
        if (wanted <= 0) return
        k = findloc(lod_segments >= wanted, .true., 1)
        if (k == 0) k = size(lod_segments)
        if (.not. c_associated(self%lod_mesh(k))) call build_lod_mesh(self, k, compile=self%shown)
    end subroutine sphere_update
