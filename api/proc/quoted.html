    function quoted(s) result(q)
        character(len=*), intent(in) :: s                   !! the text
        character(len=:), allocatable :: q                  !! the JSON string
        character(len=6) :: u
        integer :: k, c
        q = '"'
        do k = 1, len(s)
            c = iachar(s(k:k))
            select case (c)
            case (34)
                q = q//'\"'
            case (92)
                q = q//'\\'
            case (10)
                q = q//'\n'
            case (60, 0:9, 11:31)
                write(u, '(a2,z4.4)') '\u', c
                q = q//u
            case default
                q = q//s(k:k)
            end select
        end do
        q = q//'"'
    end function quoted
