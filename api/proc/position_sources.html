position_sources

public pure function position_sources(dof) result(s)

The sources of a trajectory's positions: x, y, (z) of the position, and zero for components beyond dof.

    pure function position_sources(dof) result(s)
        integer, intent(in) :: dof      !! degrees of freedom (2 or 3)
        type(DataSource) :: s(3)
        integer :: k
        do k = 1, 3
            if (k <= dof) s(k) = DataSource(SOURCE_POSOPT, element=k)
        end do
    end function position_sources
