    pure function lower(s) result(t)
        character(len=*), intent(in) :: s                   !! the text
        character(len=len(s)) :: t                          !! in lower case
        integer :: k
        t = s
        do k = 1, len(s)
            if (s(k:k) >= 'A' .and. s(k:k) <= 'Z') t(k:k) = achar(iachar(s(k:k)) + 32)
        end do
    end function lower
