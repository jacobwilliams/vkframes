public function json_number(x) result(s)

A number as JSON (17 significant digits; whole numbers without a fraction; beyond +-1e30, +-1e30, as the viewer's shaders are single precision).

    function json_number(x) result(s)
        real(wp), intent(in) :: x                           !! the number
        character(len=:), allocatable :: s                  !! its JSON text
        character(len=32) :: buf
        real(wp) :: y
        y = x
        if (y /= y) y = 0.0_wp                            ! (NaN)
        y = max(-1.0e30_wp, min(1.0e30_wp, y))
        if (y == aint(y) .and. abs(y) < 1.0e15_wp) then
            write(buf, '(i0)') int(y, int64)
        else
            write(buf, '(es25.16e3)') y
        end if
        s = trim(adjustl(buf))
    end function json_number
