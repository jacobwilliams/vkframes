    pure function point_rgba_or(traj, i, own) result(rgba)
        type(Trajectory), intent(in) :: traj
        integer, intent(in) :: i
        real(wp), intent(in) :: own(4)
        real(wp) :: rgba(4)
        rgba = own
        if (i >= 1 .and. i <= traj%num_colors()) rgba = traj%color(i)
    end function point_rgba_or
