procedure, public :: pick => window_pick

  • private subroutine window_pick(self, x, y, hits, tolerance, kinds, nearest)

    What is under point (x, y) of the window (-1 to 1 across it, y up: as the event handler's positions), nearest first (see the module documentation): none before the first frame. tolerance: how close lines, markers and frames' origins must be, in window units (logical pixels; default 4), beyond half a line's width or a marker's size. kinds: only hits of these kinds (PICK_*; the others aren't tested). nearest: only the nearest hit (cheaper: what is deeper than a hit found is skipped, e.g. to pick on every mouse move). Frames and artists can be left out (set_pickable(.false.)).

    what is there, nearest first

    window units (default 4)
    integer, intent(in), optional :: kinds(:)

procedure, public :: start_recording

  • private function start_recording(self, file, fps, quality, codec, scale, options, watermark, watermark_width, watermark_opacity, ffmpeg) result(ok)

    Start recording the frames drawn to the video file (see the module documentation), fps frames per second (default 30). Encoded by codec (an ffmpeg encoder; default libx264, H.264) at quality (for libx264, libx265 and libvpx-vp9 the CRF: 0 lossless to 51, default 18, visually lossless), scaled by scale (default 1; e.g. 0.5 for half the size of a Retina window's pixels), the size made even. options replaces the encoder options (codec, quality; and the scale, unless there is a watermark), e.g. '-c:v prores_ks -profile:v 3'. watermark: an image (e.g. a logo; PNG, SVG, ... with transparency) drawn over the video's bottom right corner by ffmpeg, watermark_width of the video's width (default 0.2), watermark_opacity opaque (default 0.6). ffmpeg: the program (default: VKFRAMES_FFMPEG, else ffmpeg on the path). Returns false if the window isn't open, is already recording or the watermark doesn't exist; ffmpeg's errors (e.g. an unknown encoder) show when it starts, at the next frame, and end the recording.

    Arguments

    Type IntentOptional Attributes Name
    class(WindowProxy), intent(inout) :: self
    character(len=*), intent(in) :: file

    the video file (e.g. .mp4, .mov, .webm)

    frames per second of the video (default 30)
    integer, intent(in), optional :: quality

    the CRF (default 18; lower is better)
    character(len=*), intent(in), optional :: codec

    an ffmpeg encoder (default libx264)
    real(kind=wp), intent(in), optional :: scale

    of the window's pixels (default 1)
    character(len=*), intent(in), optional :: options

    ffmpeg's encoder options, instead of those above
    character(len=*), intent(in), optional :: watermark

    an image file drawn at the bottom right
    real(kind=wp), intent(in), optional :: watermark_width

    its width, of the video's (default 0.2)
    real(kind=wp), intent(in), optional :: watermark_opacity

    0 to 1 (default 0.6)
    character(len=*), intent(in), optional :: ffmpeg

    type, public :: WindowProxy
        private
        type(c_ptr) :: viewer_ = c_null_ptr                 !! the rendering layer's viewer (null while closed)
        type(FrameManager), pointer :: scene => null()      !! the scene shown (not owned)
        integer :: scene_seen = -1                          !! the scene's updates at this window's last frame
        type(SimulationClock) :: clock_                     !! the window's simulation clock
        logical :: scene_added = .false.                    !! scene, sky, HUD texts given to the viewer
        real(wp) :: background(3) = [0.0_wp, 0.0_wp, 0.0_wp]  !! background color [r, g, b] (0 to 1)
        logical :: running = .false.                        !! open, and not closed by the user
        integer :: width_ = 1, height_ = 1                  !! logical size (for the views' aspect ratio)
        real(wp) :: pixel_ratio_ = 1.0_wp                   !! framebuffer pixels per window unit (a host's window)
        ! views
        type(ViewPointer), allocatable :: views(:)          !! not owned (first n_views used)
        integer :: n_views = 0, current = 0                 !! number of views; the current one (0: none)
        type(View) :: default_view                          !! used when there are no views
        type(View), pointer :: shown => null()              !! the view whose camera is in the renderer
        ! HUD text
        type(HUDEntry), allocatable :: huds(:)              !! the HUD texts shown (first n_huds used)
        integer :: n_huds = 0                               !! number of HUD texts
        type(SkySphere), pointer :: sky_ => null()          !! the background sky
        ! the window's light (OpenFrames' default: a headlight)
        real(wp) :: light_ambient_(3) = 0.2_wp, light_diffuse_(3) = 1.0_wp, light_specular_(3) = 0.0_wp  !! [r, g, b]
        real(wp) :: light_position_(4) = [0.0_wp, 0.0_wp, 1.0_wp, 0.0_wp]  !! eye coordinates (w 0: directional)
        logical :: light_on = .true.                        !! the window's light enabled (no frame's light source)
        ! keyboard and mouse
        class(WindowEventHandler), pointer :: handler => null()  !! the program's event handler (not owned)

        class(DragHandle), pointer :: dragged_handle => null()   !! the handle grabbed (not owned)

        logical :: hosted = .false.                         !! in a host's window (its size from `resize`)

        logical :: camera_throw_ = .true.                   !! the camera keeps moving after a drag released while moving

        logical :: recording_ = .false.                     !! recording (start_recording)
        real(wp) :: record_fps = 30.0_wp                    !! the video's frames per second
